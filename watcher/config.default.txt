# さなくり 会計印刷 設定  （行頭 # はコメント）
# 変更後はトレイアイコン右クリック→終了 → start_watcher.bat をダブルクリックで再起動
# このファイル(config.txt)は自動更新で上書きされません（config.default.txt が最新のひな形）

# プリンター名（Windowsの「プリンターとスキャナー」に出る名前そのまま）
# 空欄なら「通常使うプリンター」に出す。 list_printers.bat で一覧を確認できます
PRINTER=OKI B432(A5)

# 用紙サイズ（A5 / A4 / B5 など）

# 用紙の番号指定（通常は空欄でOK。PAPER から自動で決める。A4=9 A5=11 B5=13）

# 拡大縮小（shrink=大きい時だけ縮小 / fit=用紙に合わせる / noscale=等倍）
SCALE=shrink

# ダウンロードフォルダ（空欄=Windowsの既定のダウンロードフォルダ。Chromeで保存先を変えている場合のみ指定）

KEEP_DAYS=60

UPDATE_SOURCE=https://raw.githubusercontent.com/torusanada917-stack/sanakuri-print/main/watcher/
